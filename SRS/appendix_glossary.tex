\section{Glossary}

\begin{description}
\item[DRL] Deep Reinforcement Learning
\item[Federated Learning] A machine learning approach where models are trained across multiple decentralized devices without exchanging raw data
\item[MARL] Multi-Agent Reinforcement Learning
\item[SUMO] Simulation of Urban MObility, an open-source traffic simulation package
\item[YOLO] You Only Look Once, a real-time object detection system
\item[RLlib] Ray's reinforcement learning library
\item[Clustering] Grouping of intersections for federated learning coordination
\end{description}
